Attendance Marker
START

CLASS AttendanceMarker
    VARIABLE attendance[NUM_STUDENTS][NUM_DAYS]   // 2D array

    METHOD markAttendance(studentId, day, status)   // status = 'P' or 'A'
        attendance[studentId][day] = status
        PRINT "Attendance marked for Student " + studentId + " on Day " + day + " as " + status

    METHOD updateAttendance(studentId, day, status)
        attendance[studentId][day] = status
        PRINT "Attendance updated for Student " + studentId + " on Day " + day + " as " + status

    METHOD generateReport(studentId)
        presentCount = 0
        FOR day = 1 TO NUM_DAYS
            IF attendance[studentId][day] == 'P'
                presentCount = presentCount + 1
        PRINT "Student " + studentId + " was present for " + presentCount + " days in this month"


--------------------------------------

CLASS Simulation
    METHOD MAIN()
        NUM_STUDENTS = 3
        NUM_DAYS = 30
        tracker = new AttendanceMarker()

        // Sample marking for Student 1 first week
        tracker.markAttendance(1, 1, 'P')
        tracker.markAttendance(1, 2, 'P')
        tracker.markAttendance(1, 3, 'A')
        tracker.markAttendance(1, 4, 'P')
        tracker.markAttendance(1, 5, 'P')
        tracker.markAttendance(1, 6, 'P')
        tracker.markAttendance(1, 7, 'A')

        // Teacher updates Day 3
        tracker.updateAttendance(1, 3, 'P')

        // End of month report
        tracker.generateReport(1)

END
