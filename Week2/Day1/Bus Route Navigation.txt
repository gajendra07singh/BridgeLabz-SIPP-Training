Bus Route Navigation

START

CLASS BusStop
    VARIABLE name
    VARIABLE next

    CONSTRUCTOR(name)
        this.name = name
        this.next = null


--------------------------------------

CLASS BusRoute
    VARIABLE head

    CONSTRUCTOR()
        this.head = null

    METHOD addStop(stopName)
        newStop = new BusStop(stopName)
        IF head == null
            head = newStop
        ELSE
            temp = head
            WHILE temp.next != null
                temp = temp.next
            temp.next = newStop
        PRINT "Added stop: " + stopName

    METHOD insertStopAfter(targetStop, newStopName)
        temp = head
        WHILE temp != null
            IF temp.name == targetStop
                newStop = new BusStop(newStopName)
                newStop.next = temp.next
                temp.next = newStop
                PRINT "Inserted " + newStopName + " after " + targetStop
                RETURN
            temp = temp.next
        PRINT "Stop " + targetStop + " not found."

    METHOD removeStop(stopName)
        IF head == null
            PRINT "Route is empty."
            RETURN

        IF head.name == stopName
            head = head.next
            PRINT "Removed stop: " + stopName
            RETURN

        prev = head
        curr = head.next
        WHILE curr != null
            IF curr.name == stopName
                prev.next = curr.next
                PRINT "Removed stop: " + stopName
                RETURN
            prev = curr
            curr = curr.next

        PRINT "Stop " + stopName + " not found."

    METHOD searchStop(stopName)
        temp = head
        WHILE temp != null
            IF temp.name == stopName
                PRINT "Stop " + stopName + " found in the route."
                RETURN true
            temp = temp.next
        PRINT "Stop " + stopName + " not found."
        RETURN false

    METHOD displayRoute()
        temp = head
        PRINT "Bus Route: "
        WHILE temp != null
            PRINT temp.name + " -> "
            temp = temp.next
        PRINT "End of Route"


--------------------------------------

CLASS Simulation
    METHOD MAIN()
        route = new BusRoute()

        // Initial route
        route.addStop("Depot")
        route.addStop("Main Street")
        route.addStop("City Mall")
        route.addStop("Hospital")
        route.addStop("Airport")
        route.displayRoute()

        // Insert University Stop after City Mall
        route.insertStopAfter("City Mall", "University Stop")
        route.displayRoute()

        // Remove Hospital
        route.removeStop("Hospital")
        route.displayRoute()

        // Search for Airport
        route.searchStop("Airport")

END
